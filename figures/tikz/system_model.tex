% Single-column TikZ fragment for the receiver-only system model.
\begin{tikzpicture}[
    font=\scriptsize,
    >=Latex,
    block/.style={draw,rounded corners,align=center,minimum height=6.5mm,text width=30mm},
    smallblock/.style={draw,rounded corners,align=center,minimum height=6.5mm,text width=25mm},
    flow/.style={->,thick},
    note/.style={align=center,font=\scriptsize}
]

% Source and retained transmitter
\node (src) at (0,0) {Source tokens $\mathbf{s}$};
\node[block] (sem) at (0,-0.75) {Teacher semantic encoder\\$E_{\theta}$};
\node[block] (chenc) at (0,-1.65) {Teacher channel encoder\\$C_{\theta}$};
\node[draw,dashed,fit=(sem)(chenc),inner sep=2.5mm,
      label={[font=\scriptsize]right:Frozen transmitter}] (txbox) {};

% Channel path
\node[note] (x) at (0,-2.55) {Transmitted representation $\mathbf{X}$};
\node[block] (channel) at (0,-3.35) {Rayleigh channel\\+ perfect-CSI equalization};
\node[note] (z) at (0,-4.20) {Shared received representation $\mathbf{Z}$};

% Receiver branches
\node[smallblock,fill=gray!10] (teacher) at (-1.85,-5.25)
    {Teacher receiver\\8 layers / 16 heads\\Frozen};
\node[smallblock,fill=blue!6] (student) at (1.85,-5.25)
    {Student receiver\\4 layers / 8 heads\\Trainable};

\node[note] (tout) at (-1.85,-6.20) {Teacher logits / features};
\node[note] (sout) at (1.85,-6.20) {Student logits / features};

% Flow
\draw[flow] (src)--(sem);
\draw[flow] (sem)--(chenc);
\draw[flow] (chenc)--(x);
\draw[flow] (x)--(channel);
\draw[flow] (channel)--(z);
\draw[flow] (z.south)--++(0,-0.20)-|(teacher.north);
\draw[flow] (z.south)--++(0,-0.20)-|(student.north);
\draw[flow] (teacher)--(tout);
\draw[flow] (student)--(sout);

\end{tikzpicture}
